rails new . --database=postgresql

# Generated Dockerfile is production-oriented Rails image, let's create Dockerfile.dev for a fast local dev image
# And docker-compose.yml for local infrastructure

# Dockerfile is how to build/run one Rails container
# docker-compose.yml is how multiple containers work together

# Rails doesn't assume we'll use docker compose in production, so that's not generated


####### USEFUL DOCKER COMMANDS #######

# build the containers
docker compose build

# Start everything
docker compose up

# Stop everything
docker compose down

# see Docker processes
docker ps

# Kill every Docker process
docker kill $(docker ps -q)

# Rails commands
docker compose exec web bin/rails (console | db:migrate | test)

# Create dev and test databases
docker compose run --rm web bin/rails db:create

# Open PostgreSQL
docker compose exec db psql \
  -U postgres \
  -d appointments_practice_development

# Check services logs
docker compose logs -f (web | db | sidekiq | redis)

######################################

# Create models

docker compose exec web bin/rails g model Account name:string active:boolean
docker compose exec web bin/rails g model Customer account:references name:string email:string active:boolean
docker compose exec web bin/rails g model Appointment account:references customer:references \
  starts_at:datetime status:string price_cents:integer

# Before migrating, inspect and update migrations
# Critical integrity rules should live in PostgreSQL, not only in Rails validations.
# Don't add indexes blindly, Rails already adds indexes for foreign keys

# May need to change ownership of Rails-generated files from root to user
sudo chown -R $USER:$USER .

# Update generated models to add associations and validations
# Run migrations (it creates schema.rb the first time)
docker compose exec web bin/rails db:migrate

# Create some records in bin/rails console
account = Account.create!(name: 'Test Salon')
customer = account.customers.create!(name: 'John Doe', email: 'john@example.com')
appointment = account.appointments.create!(customer: customer, starts_at: 1.day.from_now, price_cents: 25_000)

# Installing Sidekiq & Redis
# Add Redis and Sidekiq services to docker-compose.yml
# Add sidekiq gem # Automatically installs redis as a dependency

# Install new gems with
docker compose run --rm web bundle install

# docker compose run --rm is the universal, fail-safe choice for dependency updates, 
# because it works regardless of whether your app containers are running or crashed
# docker compose run creates a fresh, temporary container to run the command. --rm ensures it gets deleted as soon as the task finishes
# docker compose exec runs inside already running container (no new container is created, nothing to remove)

# And build and start again with latest Gemfile
docker compose build
docker compose up

# Check that now we have a container for: web, postgres, sidekiq and redis
docker ps
